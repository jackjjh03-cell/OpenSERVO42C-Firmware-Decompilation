cmake_minimum_required(VERSION 3.20)
project(SERVO42C LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_C_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_C_COMPILER_TARGET STREQUAL "arm-none-eabi")
    message(FATAL_ERROR "Configure using -DCMAKE_TOOLCHAIN_FILE=cmake/arm-clang.cmake")
endif()
if(CMAKE_BUILD_TYPE)
    message(FATAL_ERROR "Leave CMAKE_BUILD_TYPE empty to retain the release compiler flags")
endif()
option(SYSINFO_PROFILE "Experimental motion IRQ profiling" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(VARIANT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/factory")
file(GLOB SOURCES CONFIGURE_DEPENDS "${VARIANT_DIR}/src/*.c")
list(SORT SOURCES)
add_executable(servo42c ${SOURCES})
set_target_properties(servo42c PROPERTIES SUFFIX ".elf" LINK_DEPENDS "${VARIANT_DIR}/layout.ld")
if(SYSINFO_PROFILE)
    target_compile_definitions(servo42c PRIVATE SYSINFO_PROFILE=1)
endif()
target_include_directories(servo42c PRIVATE "${VARIANT_DIR}/include")
target_compile_options(servo42c PRIVATE -mcpu=cortex-m0plus -mthumb -Oz -fomit-frame-pointer -ffunction-sections -fdata-sections -ffreestanding -fno-builtin -fno-strict-aliasing -Wall -Wextra -Werror)
set(CMAKE_C_LINK_EXECUTABLE "\"${CMAKE_LINKER}\" <LINK_FLAGS> -o <TARGET> <OBJECTS>")
target_link_options(servo42c PRIVATE --gc-sections --icf=safe --no-undefined "-T${VARIANT_DIR}/layout.ld" "-Map=${CMAKE_CURRENT_BINARY_DIR}/servo42c.map")
add_custom_command(TARGET servo42c POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/inspect_elf.py" "$<TARGET_FILE:servo42c>" --report "${CMAKE_CURRENT_BINARY_DIR}/build_report.json" VERBATIM)
set(SYSINFO_TEST_ARGS)
if(SYSINFO_PROFILE)
    list(APPEND SYSINFO_TEST_ARGS --profile)
endif()
add_custom_target(verify
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_sysinfo.py" "$<TARGET_FILE:servo42c>" ${SYSINFO_TEST_ARGS}
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_config_editor.py"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" DEPENDS servo42c VERBATIM)
